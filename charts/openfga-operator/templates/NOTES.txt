  kubectl get configmap -n {{ include "openfga-operator.watchNamespace" . }} -l app.kubernetes.io/managed-by=openfga-operator

  kubectl get jobs -n {{ include "openfga-operator.watchNamespace" . }} -l app.kubernetes.io/part-of=openfga,app.kubernetes.io/component=migration
